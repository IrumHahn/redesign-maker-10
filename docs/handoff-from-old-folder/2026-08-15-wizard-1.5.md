```tsx
import React, { useState } from 'react';

interface Props {
  value: string;
  onChange: (v: string) => void;
  className?: string;
  multiline?: boolean;
  placeholder?: string;
}

export const EditableText: React.FC<Props> = ({ value, onChange, className = '', multiline = false, placeholder }) => {
  const [editing, setEditing] = useState(false);
  const [draft, setDraft] = useState(value);

  const commit = () => { onChange(draft.trim()); setEditing(false); };

  if (!editing) {
    return (
      <span
        className={`${className} cursor-text hover:bg-yellow-50 hover:outline hover:outline-1 hover:outline-yellow-300 rounded px-0.5 -mx-0.5 block`}
        title="클릭해서 수정"
        onClick={() => { setDraft(value); setEditing(true); }}
      >
        {value || <span className="text-gray-300">{placeholder ?? '클릭해서 입력'}</span>}
      </span>
    );
  }
  const shared = {
    autoFocus: true,
    value: draft,
    onChange: (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => setDraft(e.target.value),
    onBlur: commit,
    className: `${className} w-full border border-blue-300 rounded px-1 focus:outline-none focus:ring-1 focus:ring-blue-400 bg-white`,
  };
  return multiline
    ? <textarea {...shared} rows={2} onKeyDown={e => { if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); commit(); } }} />
    : <input {...shared} onKeyDown={e => { if (e.key === 'Enter') commit(); }} />;
};
```

```ts
export async function regenerateSectionCopy(
  apiKey: string,
  section: SectionBlueprint,
  additionalInfo?: string,
  desiredTone?: string
): Promise<Pick<SectionBlueprint, 'headline' | 'subheadline' | 'bullets' | 'trust_or_objection_line' | 'CTA'>> {
  const ai = new GoogleGenAI({ apiKey });
  const response = await ai.models.generateContent({
    model: 'gemini-3-flash-preview',
    contents: {
      parts: [{
        text: `한국 이커머스 상세페이지의 "${section.section_name}" 섹션(역할: ${section.goal}) 카피를 새로 제안해주세요.
현재 카피 — 헤드라인: ${section.headline} / 서브헤드: ${section.subheadline} / 불릿: ${section.bullets?.join(', ')}
${additionalInfo ? `[제품 추가 정보]: ${additionalInfo}` : ''}
${desiredTone ? `[톤]: ${desiredTone}` : ''}
규칙: 현재 카피와 다른 각도로, 더 구체적이고 팔리는 문장으로. 확인되지 않는 수치/인증/효능은 만들지 말 것. 불릿은 3개.`
      }]
    },
    config: {
      responseMimeType: 'application/json',
      responseSchema: {
        type: Type.OBJECT,
        properties: {
          headline: { type: Type.STRING },
          subheadline: { type: Type.STRING },
          bullets: { type: Type.ARRAY, items: { type: Type.STRING } },
          trust_or_objection_line: { type: Type.STRING },
          CTA: { type: Type.STRING },
        }
      }
    }
  });
  if (!response.text) throw new Error("카피 재생성 응답이 비어 있습니다.");
  return JSON.parse(response.text);
}
```

```ts
export type OverlayTemplate = 'hero' | 'info' | 'cta' | 'none';

export interface OverlayCopy {
  headline: string;
  subheadline?: string;
  bullets?: string[];
  cta?: string;
}

export interface TonePreset {
  accent: string;   // 배지/칩 배경
  scrim: string;    // 히어로 하단 그라데이션 시작색 (rgba)
  text: string;     // 본문 텍스트
  chipText: string;
}

export const TONE_PRESETS: Record<string, TonePreset> = {
  default:   { accent: '#2563EB', scrim: 'rgba(17,24,39,0.82)', text: '#FFFFFF', chipText: '#FFFFFF' },
  '프리미엄': { accent: '#B48A3C', scrim: 'rgba(20,16,8,0.85)',  text: '#F8F5EF', chipText: '#1F1B10' },
  '모던':    { accent: '#111827', scrim: 'rgba(17,24,39,0.82)', text: '#FFFFFF', chipText: '#FFFFFF' },
  '테크':    { accent: '#0EA5E9', scrim: 'rgba(2,6,23,0.85)',   text: '#E0F2FE', chipText: '#03263B' },
  '미니멀':  { accent: '#374151', scrim: 'rgba(255,255,255,0.86)', text: '#111827', chipText: '#FFFFFF' },
  '팝아트':  { accent: '#F43F5E', scrim: 'rgba(30,10,20,0.82)', text: '#FFF1F2', chipText: '#FFFFFF' },
  '인스타감성': { accent: '#EC4899', scrim: 'rgba(40,20,35,0.78)', text: '#FDF2F8', chipText: '#FFFFFF' },
  '레트로':  { accent: '#D97706', scrim: 'rgba(41,25,4,0.85)',  text: '#FEF3C7', chipText: '#3B2A05' },
};

export function resolveTone(tone?: string): TonePreset {
  return (tone && TONE_PRESETS[tone]) || TONE_PRESETS['default'];
}

export function inferTemplate(sectionName: string, index: number, total: number): OverlayTemplate {
  if (index === 0) return 'hero';
  if (/cta|오퍼|구매|혜택|마감/i.test(sectionName) || index === total - 1) return 'cta';
  return 'info';
}

const FONT = '"Pretendard Variable", Pretendard, "Noto Sans KR", sans-serif';

async function loadImage(src: string): Promise<HTMLImageElement> {
  return new Promise((resolve, reject) => {
    const img = new Image();
    img.onload = () => resolve(img);
    img.onerror = reject;
    img.src = src;
  });
}

/** 줄바꿈: maxWidth를 넘지 않게 어절 단위로 자름 */
function wrapText(ctx: CanvasRenderingContext2D, text: string, maxWidth: number): string[] {
  const words = text.split(/\s+/);
  const lines: string[] = [];
  let line = '';
  for (const w of words) {
    const test = line ? `${line} ${w}` : w;
    if (ctx.measureText(test).width > maxWidth && line) { lines.push(line); line = w; }
    else line = test;
  }
  if (line) lines.push(line);
  return lines;
}

/** base 이미지(dataURL) 위에 카피를 합성해 PNG dataURL 반환 */
export async function renderOverlay(
  baseImage: string,
  template: OverlayTemplate,
  copy: OverlayCopy,
  tone?: string
): Promise<string> {
  const img = await loadImage(baseImage);
  const W = img.naturalWidth, H = img.naturalHeight;
  const canvas = document.createElement('canvas');
  canvas.width = W; canvas.height = H;
  const ctx = canvas.getContext('2d');
  if (!ctx) throw new Error('Canvas 컨텍스트 생성 실패');
  const t = resolveTone(tone);
  try { await (document as any).fonts?.load(`700 ${Math.round(W * 0.05)}px ${FONT}`); } catch { /* 폰트 미로드 시 폴백 */ }

  ctx.drawImage(img, 0, 0);
  if (template === 'none') return canvas.toDataURL('image/png');

  const pad = W * 0.06;
  const headlineSize = Math.round(W * (template === 'hero' ? 0.062 : 0.05));
  const subSize = Math.round(W * 0.032);
  const chipSize = Math.round(W * 0.028);

  if (template === 'hero' || template === 'cta') {
    // 하단 스크림
    const scrimH = H * 0.42;
    const grad = ctx.createLinearGradient(0, H - scrimH, 0, H);
    grad.addColorStop(0, 'rgba(0,0,0,0)');
    grad.addColorStop(0.45, t.scrim);
    grad.addColorStop(1, t.scrim);
    ctx.fillStyle = grad;
    ctx.fillRect(0, H - scrimH, W, scrimH);

    ctx.textAlign = template === 'cta' ? 'center' : 'left';
    const x = template === 'cta' ? W / 2 : pad;
    let y = H - pad;

    // CTA 배지 (cta 템플릿, copy.cta 존재 시)
    if (template === 'cta' && copy.cta) {
      ctx.font = `700 ${chipSize}px ${FONT}`;
      const cw = ctx.measureText(copy.cta).width + chipSize * 2;
      const ch = chipSize * 2.2;
      ctx.fillStyle = t.accent;
      ctx.beginPath();
      ctx.roundRect(W / 2 - cw / 2, y - ch, cw, ch, ch / 2);
      ctx.fill();
      ctx.fillStyle = t.chipText;
      ctx.textBaseline = 'middle';
      ctx.fillText(copy.cta, W / 2, y - ch / 2);
      ctx.textBaseline = 'alphabetic';
      y -= ch + subSize;
    }
    // 서브헤드
    if (copy.subheadline) {
      ctx.font = `500 ${subSize}px ${FONT}`;
      ctx.fillStyle = t.text;
      ctx.globalAlpha = 0.9;
      const lines = wrapText(ctx, copy.subheadline, W - pad * 2).slice(0, 2);
      for (const line of [...lines].reverse()) { ctx.fillText(line, x, y); y -= subSize * 1.5; }
      ctx.globalAlpha = 1;
      y -= subSize * 0.5;
    }
    // 헤드라인 (아래에서 위로 쌓기)
    ctx.font = `800 ${headlineSize}px ${FONT}`;
    ctx.fillStyle = t.text;
    const hLines = wrapText(ctx, copy.headline, W - pad * 2).slice(0, 3);
    for (const line of [...hLines].reverse()) { ctx.fillText(line, x, y); y -= headlineSize * 1.28; }
  } else {
    // info: 상단 헤드라인 + 불릿 칩
    const topH = H * 0.34;
    const grad = ctx.createLinearGradient(0, 0, 0, topH);
    grad.addColorStop(0, t.scrim);
    grad.addColorStop(0.6, t.scrim);
    grad.addColorStop(1, 'rgba(0,0,0,0)');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, topH);

    ctx.textAlign = 'left';
    let y = pad + headlineSize;
    ctx.font = `800 ${headlineSize}px ${FONT}`;
    ctx.fillStyle = t.text;
    for (const line of wrapText(ctx, copy.headline, W - pad * 2).slice(0, 2)) {
      ctx.fillText(line, pad, y); y += headlineSize * 1.28;
    }
    if (copy.subheadline) {
      ctx.font = `500 ${subSize}px ${FONT}`;
      ctx.globalAlpha = 0.9;
      ctx.fillText(wrapText(ctx, copy.subheadline, W - pad * 2)[0] ?? '', pad, y);
      ctx.globalAlpha = 1;
      y += subSize * 1.6;
    }
    // 불릿 칩 최대 3개
    if (copy.bullets?.length) {
      ctx.font = `600 ${chipSize}px ${FONT}`;
      let cx = pad;
      for (const b of copy.bullets.slice(0, 3)) {
        const cw = ctx.measureText(b).width + chipSize * 1.6;
        const ch = chipSize * 2;
        if (cx + cw > W - pad) { cx = pad; y += ch + chipSize * 0.6; }
        ctx.fillStyle = t.accent;
        ctx.beginPath();
        ctx.roundRect(cx, y - ch * 0.7, cw, ch, ch / 2);
        ctx.fill();
        ctx.fillStyle = t.chipText;
        ctx.textBaseline = 'middle';
        ctx.fillText(b, cx + chipSize * 0.8, y - ch * 0.7 + ch / 2);
        ctx.textBaseline = 'alphabetic';
        cx += cw + chipSize * 0.6;
      }
    }
  }
  return canvas.toDataURL('image/png');
}
```

```ts
import type { LandingPageBlueprint } from '../types';

export function blueprintToMarkdown(bp: LandingPageBlueprint): string {
  const lines: string[] = ['# 상세페이지 블루프린트', ''];
  if (bp.executiveSummary) lines.push('## 요약', bp.executiveSummary, '');
  bp.sections?.forEach((s, i) => {
    lines.push(`## ${i + 1}. ${s.section_name}`, `**역할:** ${s.goal}`, '',
      `**헤드라인:** ${s.headline}`, `**서브헤드:** ${s.subheadline}`, '');
    if (s.bullets?.length) { lines.push('**핵심 포인트:**', ...s.bullets.map(b => `- ${b}`), ''); }
    if (s.trust_or_objection_line) lines.push(`**신뢰 문구:** ${s.trust_or_objection_line}`, '');
    if (s.CTA) lines.push(`**CTA:** ${s.CTA}`, '');
  });
  return lines.join('\n');
}
```

```ts
import JSZip from 'jszip';
import type { SectionBlueprint } from '../types';

export function buildZipFileName(index: number, sectionName: string): string {
  const safe = sectionName.replace(/[\\/:*?"<>|]/g, '_').replace(/\s+/g, '');
  return `${String(index + 1).padStart(2, '0')}_${safe}.png`;
}

/** composedImage 우선, 없으면 generatedImage. 이미지 없는 섹션은 건너뜀. */
export async function exportSectionsAsZip(sections: SectionBlueprint[]): Promise<Blob> {
  const zip = new JSZip();
  sections.forEach((s, i) => {
    const src = s.composedImage ?? s.generatedImage;
    if (!src) return;
    zip.file(buildZipFileName(i, s.section_name), src.split(',')[1], { base64: true });
  });
  return zip.generateAsync({ type: 'blob' });
}

export function downloadBlob(blob: Blob, fileName: string) {
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url; a.download = fileName; a.click();
  URL.revokeObjectURL(url);
}
```

```ts
import type { GeneratedResult, AspectRatio } from '../types';

export interface WizardSession {
  result: GeneratedResult;
  aspectRatio: AspectRatio;
  desiredTone: string;
  savedAt?: number;
}

const KEY = 'wizardSession_v1';

export function serializeSession(s: Omit<WizardSession, 'savedAt'>): string {
  const slim: WizardSession = {
    ...s,
    savedAt: Date.now(),
    result: {
      ...s.result,
      blueprint: {
        ...s.result.blueprint,
        sections: s.result.blueprint.sections.map(({ generatedImage, composedImage, ...rest }) => rest),
      },
    },
  };
  return JSON.stringify(slim);
}

export function deserializeSession(raw: string | null): WizardSession | null {
  if (!raw) return null;
  try { return JSON.parse(raw) as WizardSession; } catch { return null; }
}

export function saveSession(s: Omit<WizardSession, 'savedAt'>) {
  try { localStorage.setItem(KEY, serializeSession(s)); } catch { /* 용량 초과 시 무시 */ }
}
export function loadSession(): WizardSession | null {
  return deserializeSession(localStorage.getItem(KEY));
}
export function clearSession() { localStorage.removeItem(KEY); }
```
